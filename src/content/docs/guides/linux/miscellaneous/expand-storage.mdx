<Steps>

1.  ## Extend the logical volume

    <Code lang="sh" code={`sudo lvextend -l +100%FREE /dev/ubuntu-vg/ubuntu-lv`} />

2.  ## Resize the filesystem

    <Code lang="sh" code={`sudo resize2fs /dev/ubuntu-vg/ubuntu-lv`} />

3.  ## Verify

    <Code lang="sh" code={`df -h /`} />

</Steps>
